const Option: React.FC<SidebarOptionProps> = ({ Icon, title, selected, setSelected, open, notifs, onClick }) => {
  const isSelected = selected === title.toLowerCase().replace(/\s+/g, '-');
  
  return (
    <button
      onClick={() => {
        if (onClick) {
          onClick();
        } else {
          setSelected(title.toLowerCase().replace(/\s+/g, '-'));
        }
      }}
      className={`relative flex h-11 w-full items-center rounded-md transition-all duration-200 ${
        isSelected 
          ? "bg-gray-100 dark:bg-gray-800 text-gray-900 dark:text-gray-100 shadow-sm border-l-2 border-gray-500 dark:border-gray-400" 
          : "text-gray-600 dark:text-gray-400 hover:bg-gray-50 dark:hover:bg-gray-800 hover:text-gray-900 dark:hover:text-gray-200"
      }`}
    >
      <div className="grid h-full w-12 place-content-center">
        <Icon className="h-4 w-4" />
      </div>
      
      {open && (
        <span
          className={`text-sm font-medium transition-opacity duration-200 ${
            open ? 'opacity-100' : 'opacity-0'
          }`}
        >
          {title}
        </span>
      )}

      {notifs && open && (
        <span className="absolute right-3 flex h-5 w-5 items-center justify-center rounded-full bg-gray-500 dark:bg-gray-600 text-xs text-white font-medium">
          {notifs}
        </span>
      )}
    </button>
  );
};

        <button className="flex items-center gap-2 px-4 py-2 bg-white dark:bg-gray-900 border border-gray-200 dark:border-gray-800 rounded-lg text-sm font-medium text-gray-700 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-800">
          <Download className="h-4 w-4" />
          Export History
        </button>
      </div>

      <div className="rounded-xl border border-gray-200 dark:border-gray-800 bg-white dark:bg-gray-900 shadow-sm">
        <div className="p-6 border-b border-gray-200 dark:border-gray-700">
          <h3 className="text-lg font-semibold text-gray-900 dark:text-gray-100">Activity History</h3>
          <p className="text-sm text-gray-500 dark:text-gray-400 mt-1">Complete record of all your activities</p>
        </div>
        
        <div className="divide-y divide-gray-200 dark:divide-gray-700">
          {mockHistory.map((item) => (
            <div key={item.id} className="p-6 hover:bg-gray-50 dark:hover:bg-gray-800">
              <div className="flex items-start justify-between">
                <div className="flex items-start gap-4">
                  <div className={`p-2 rounded-lg ${
                    item.type === 'transfer' ? 'bg-gray-100 dark:bg-gray-800' :
                    item.type === 'analysis' ? 'bg-gray-100 dark:bg-gray-800' :
                    item.type === 'monitor' ? 'bg-gray-100 dark:bg-gray-800' :
                    'bg-gray-100 dark:bg-gray-800'
                  }`}>
                    {item.type === 'transfer' && <ArrowUpDown className="h-4 w-4 text-gray-600 dark:text-gray-400" />}
                    {item.type === 'analysis' && <Search className="h-4 w-4 text-gray-600 dark:text-gray-400" />}
                    {item.type === 'monitor' && <Monitor className="h-4 w-4 text-gray-600 dark:text-gray-400" />}
                    {item.type === 'balance' && <Wallet className="h-4 w-4 text-gray-600 dark:text-gray-400" />}
                  </div>
                  <div>
                    <p className="font-medium text-gray-900 dark:text-gray-100">{item.description}</p>
                    <p className="text-sm text-gray-500 dark:text-gray-400 mt-1">{item.timestamp}</p>
                  </div>
                </div>
                <div className="flex items-center gap-2">
                  <span className={`inline-flex px-2 py-1 text-xs font-medium rounded-full ${
                    item.status === 'success' ? 'bg-gray-100 dark:bg-gray-800 text-gray-800 dark:text-gray-200' :
                    item.status === 'error' ? 'bg-gray-100 dark:bg-gray-800 text-gray-600 dark:text-gray-400' :
                    'bg-gray-100 dark:bg-gray-800 text-gray-500 dark:text-gray-500'
                  }`}>
                    {item.status}
                  </span>
                  {item.status === 'success' && <CheckCircle className="h-4 w-4 text-gray-500" />}
                  {item.status === 'error' && <AlertCircle className="h-4 w-4 text-gray-400" />}
                </div>
              </div>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
};

export default Component;
